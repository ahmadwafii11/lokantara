- pnpm init
- pnpm add express cors dotenv @prisma/client
- pnpm add -D typescript tsx prisma @types/node @types/express
- pnpm approve-builds
- ctrl + a --> y
- pnpm install
- pnpm prisma init
- pnpm exec prisma -v (cek version prisma)
- pnpm prisma generate (generate prisma)
- pnpm prisma migrate dev --name init (migrate prisma)
- pnpm prisma studio (open prisma studio localhost)
- pnpm add -D @types/cors
- pnpm dev
